John the Ripper Commands Cheat Sheet - Security Assignment

John The Ripper — Complete Command Reference Educational Use Only
1 — Basic Usage
john /etc/passwd
# Default mode par password crack karo
2 — Wordlist / Dictionary Attack
john --wordlist=/usr/share/wordlists/rockyou.txt hash.txt
# Wordlist (dictionary) se attack karo
3 — Hash Format Specify karna
john --format=md5crypt hash.txt
john --format=sha512crypt hash.txt
john --format=NT hash.txt
# Hash type manually set karo (MD5, SHA512, NTLM)
4 — Cracked Passwords Dekhna
john --show hash.txt
john --show --format=md5crypt hash.txt
# Jo passwords crack hue hain unhe display karo
5 — Incremental Mode (Brute Force)
john --incremental hash.txt
john --incremental=Digits hash.txt
# Brute force — har combination try karo
6 — Rules ke saath Wordlist
john --wordlist=rockyou.txt --rules hash.txt
john --wordlist=rockyou.txt --rules=Jumbo hash.txt
# Mangling rules apply karo (uppercase, l33t, etc.)
7 — /etc/shadow ke saath use karna
unshadow /etc/passwd /etc/shadow > combined.txt
john combined.txt
# passwd + shadow combine karo phir crack karo
8 — Supported Formats List karna
john --list=formats
john --list=subformats
# Saare supported hash formats dekho
9 — Session Save & Resume
john --session=mysession hash.txt
john --restore=mysession
# Session save karo aur baad mein resume karo
10 — Multiple CPU Cores
john --fork=4 hash.txt
john --node=1-2/3 hash.txt
# Multiple cores ya nodes use karo speed ke liye
11 — zip2john / pdf2john (File Hashes)
zip2john file.zip > zip_hash.txt
john zip_hash.txt
pdf2john file.pdf > pdf_hash.txt
# ZIP/PDF password crack karo
12 — Help & Version
john --help
john --version
# Help aur version info dekho
